# -----------------------------
# R E P O  /  P P A  (third-party apt sources)
# Applied before any apt install, so the 'apt' lines below can depend on them.
# Written as deb822 .sources + a dearmored keyring under /etc/apt/keyrings.
#   repo <name> url=<URI> suite=<suite> comps=<components> key=<keyurl> [arch=<arch>]
#   pin  <name> pkg=<glob> priority=<n> [origin=<host>] [release=<expr>]
#   ppa  <owner/name>
# -----------------------------
repo 1password url=https://downloads.1password.com/linux/debian/amd64 suite=stable comps=main key=https://downloads.1password.com/linux/keys/1password.asc arch=amd64
# gh ships a much newer CLI than the Ubuntu archive (2.45.0 on noble).
repo github-cli url=https://cli.github.com/packages suite=stable comps=main key=https://cli.github.com/packages/githubcli-archive-keyring.gpg arch=amd64
# .NET SDK newer than the archive provides.
ppa dotnet/backports
# Brave, from Brave's own repo rather than Flathub or snap: 1Password will not
# connect to a sandboxed browser, and losing that costs unlock-with-system-auth
# and desktop-backed autofill. The repo serves arm64 too, but arch is pinned to
# match the other entries here and keep apt from indexing a slice we never use.
repo brave url=https://brave-browser-apt-release.s3.brave.com/ suite=stable comps=main key=https://brave-browser-apt-release.s3.brave.com/brave-browser-archive-keyring.gpg arch=amd64
# Firefox from Mozilla, not the Ubuntu archive: 'firefox' there is a stub whose
# only job is to install the snap, and 1Password refuses to talk to a sandboxed
# browser -- its native-messaging host sits in ~/.mozilla, which the snap cannot
# read. The two pins below are what actually make the deb win.
repo mozilla url=https://packages.mozilla.org/apt suite=mozilla comps=main key=https://packages.mozilla.org/apt/repo-signing-key.gpg arch=amd64
pin mozilla pkg=* origin=packages.mozilla.org priority=1000
# The stub is 1:1snap1-0ubuntu5 and that epoch sorts above Mozilla's 154.x, so
# the priority pin alone loses on any machine where the stub is preinstalled.
# -1 makes it uninstallable outright, including across release upgrades.
pin firefox-nosnap pkg=firefox release=o=Ubuntu priority=-1
# VS Code from Microsoft's repo rather than the snap. The snap is classic, so
# this buys no sandbox escape -- it buys startup (no squashfs decompress on
# cold launch), 'code' on PATH without the /snap/bin shim, and updates that
# ride ordinary apt instead of a background refresh that nags to restart.
# Settings survive the switch untouched: the classic snap already wrote to the
# real ~/.config/Code and ~/.vscode.
# Naming this repo 'vscode' is deliberate -- the deb's postinst registers the
# same repo itself unless it finds an existing source pointing at
# packages.microsoft.com/repos/code, and it writes to vscode.sources. Matching
# both the name and the URI exactly is what keeps apt from listing it twice.
repo vscode url=https://packages.microsoft.com/repos/code suite=stable comps=main key=https://packages.microsoft.com/keys/microsoft.asc arch=amd64
# Spotify from Spotify's repo rather than the snap. This one is a real win:
# the snap is strictly confined, so cold start pays AppArmor + portal setup on
# top of squashfs, and playing local files needs a manual 'snap connect'.
# Upstream ships the same version the snap does, so nothing is given up.
# Spotify rotates this signing key every few years and publishes each one under
# its own long key id, so the URL below goes stale rather than redirecting. When
# apt reports NO_PUBKEY <id> for this repo, the fix is to point this line at
# https://download.spotify.com/debian/pubkey_<id>.gpg; install_repo tracks the
# key URL and refetches on its own from there.
repo spotify url=https://repository.spotify.com suite=stable comps=non-free key=https://download.spotify.com/debian/pubkey_5384CE82BA52C83A.gpg arch=amd64

# Bevy / Rust game development. This is the machine's list rather than any one
# project's, so it is the full upstream recipe: a game that never enables
# gamepads does not link libudev, but the next one will, and a missing -dev
# package surfaces as a pkg-config failure three minutes into a cold build.
#   libasound2-dev  ALSA, reached via bevy_audio -> rodio -> cpal -> alsa-sys.
#                   The one here that is a real link rather than a dlopen.
#   libudev-dev     bevy_gilrs (gamepads).
#   libwayland-dev  } Ubuntu's session is Wayland; winit reaches it through
#   libxkbcommon-dev} smithay-client-toolkit.
# Not listed on purpose: libx11-dev and the Xlib -dev family. winit's X11 path
# is x11-dl, which dlopens at runtime, so the headers buy nothing.
apt libasound2-dev
apt libudev-dev
apt libwayland-dev
apt libxkbcommon-dev
# Linking dominates a Bevy incremental rebuild -- it is a large binary relinked
# on every one-line change -- and mold is several times faster than GNU ld at
# it. clang is what drives it (-fuse-ld=mold); a project opts in per target in
# its own .cargo/config.toml, so installing these changes nothing by itself.
apt clang
apt mold
# lld is a second linker and is here for cargo-xwin rather than for speed: a
# Windows build cross-compiled from this machine is linked by lld-link, which
# ships in this package.
apt lld
# Diagnosing a black window: 'vulkaninfo' says whether the driver, the ICD and
# the loader agree, which is the first question when a wgpu app will not start.
apt vulkan-tools

# Android. The JDK is a version rather than 'default-jdk' on purpose: AGP 9.1
# requires exactly 17 and this machine's default java is 21, so a build run
# without JAVA_HOME pointing at the 17 install fails inside a Gradle daemon that
# will not start, which reports as neither. zsh/.zshrc sets it; this is what it
# points at.
apt openjdk-17-jdk
# The udev rules that let a plugged-in phone be seen by adb as this user rather
# than only as root. Not on the critical path for wireless debugging, and one
# package is cheaper than diagnosing 'no permissions' on the day a cable is used.
apt android-sdk-platform-tools-common

# D E V  (language toolchains via official installers)
# Installed by install_linux.sh, not apt — keeps versions current and
# self-updating. PATH wiring lives in zsh/.zshrc. Safe to re-run.
#   dev rust [target...] -> rustup (~/.cargo), plus any extra rustup
#                targets named on the line (the host target is implicit)
#   dev msvc-cross -> clang-cl/llvm-lib/llvm-rc symlinks for cargo-xwin
#   dev android-sdk -> Google's command-line tools + platform/build-tools/NDK
#                (~/Android/Sdk); needs 'apt openjdk-17-jdk' above
#   dev go     -> latest official tarball (/usr/local/go)
#   dev node   -> nvm + Node LTS (~/.nvm)
#   dev neovim -> latest stable tarball (/opt/nvim, linked into /usr/local/bin)
#   dev claude -> Claude Code via official install.sh (~/.local/bin)
#   dev wezterm-> nightly .deb from GitHub releases (skipped on WSL)
#   dev lazygit-> upstream release tarball (/usr/local/bin)
#   dev glow   -> upstream .deb (stays dpkg-managed)
#   dev rustdesk-> upstream .deb (stays dpkg-managed, skipped on WSL);
#                  x86_64 gets the Wayland unattended-access preview build
# -----------------------------
# wasm32 is here rather than left to each project because it is a property of
# the machine: without it `trunk build` stops with a rustup hint, and every
# Bevy project that ships to itch.io or the web wants exactly this target.
dev rust wasm32-unknown-unknown x86_64-pc-windows-msvc aarch64-linux-android
# What cargo-xwin (below) needs on PATH to link a Windows binary. Cheap --
# three symlinks into ~/.local/bin -- and the failure without it is a cc-rs
# error about a missing 'llvm-lib' partway through a four-minute build.
dev msvc-cross
# The Android SDK and NDK, without Android Studio: the whole build loop is
# command line. The NDK version is pinned in the installer and has to match what
# a project builds against elsewhere -- see the comment on install_android_sdk.
# Large: the NDK alone is some 2.5 GB.
dev android-sdk
dev go
dev node
dev neovim
dev claude
# WezTerm is not in the Ubuntu archive at all, and its apt repo still serves the
# Feb 2024 stable tag, so this tracks the daily-rebuilt nightly instead. The
# Flatpak build is sandboxed and there is no official snap.
dev wezterm
# Neither of these is in the Ubuntu archive on any current release, so they come
# from upstream releases rather than apt.
dev lazygit
dev glow
# Remote desktop client, matching 'winget RustDesk.RustDesk' on the Windows
# side. Not in the Ubuntu archive; upstream .deb from GitHub releases. Stable
# RustDesk cannot do unattended access on Wayland (Ubuntu's default session),
# so on x86_64 this tracks upstream's preview build from the rolling 'nightly'
# tag: https://rustdesk.com/blog/unattended-remote-access-wayland/
dev rustdesk
# Also on Windows as 'winget Wilfred.difftastic'. Taken from the upstream
# release rather than the community snap, which trails a version behind.
dev difftastic
# Chrome is NOT a 'repo' entry: its package rewrites its own .sources file from
# /etc/cron.daily/google-chrome, so declaring it here too would duplicate it.
dev chrome

# -----------------------------
# C A R G O  (Rust binaries from crates.io)
# Installed with `cargo install --locked`, which is what makes them
# reproducible: without it cargo re-resolves the crate's whole dependency tree
# against today's registry, so the same version number builds differently on
# two machines a week apart. Skipped when the binary is already on PATH --
# these are slow to build and none of them self-updates, so re-running the
# installer is not the place to notice a new release.
#   cargo <crate> [--flag ...]   ->  cargo install --locked <crate> [flags]
# -----------------------------
# Trunk drives the wasm build: it compiles the crate to wasm32, runs
# wasm-bindgen and wasm-opt, and copies the asset directories the game fetches
# at runtime. It downloads its own wasm-bindgen-cli and binaryen, so this is
# the whole of the web toolchain.
cargo trunk
# Cross-compiling a Windows release from this machine. cargo-xwin fetches
# Microsoft's own CRT and SDK headers and links with clang-cl/lld-link, so what
# comes out is a genuine x86_64-pc-windows-msvc binary rather than a MinGW one
# -- which is the ABI a Steam build is normally expected to be.
cargo cargo-xwin
# The Android half of the same idea. cargo-ndk picks the NDK's clang and sysroot
# for a target triple and an API level, and stages the resulting .so where Gradle
# expects it; without it the linker invocation has to be assembled by hand.
cargo cargo-ndk
